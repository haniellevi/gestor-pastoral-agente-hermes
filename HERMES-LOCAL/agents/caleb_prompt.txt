```python
import os

OBSIDIAN_VAULT = r"C:\Users\hanie\Searches\OneDrive\Documentos\Obsidian Vault"

def ler_conhecimento(arquivo):
    if not ("/" in arquivo or "\\" in arquivo):
        filepath = f"conhecimento/{arquivo}"
        if not filepath.endswith(".md"):
            filepath += ".md"
    else:
        filepath = os.path.join(OBSIDIAN_VAULT, arquivo)
        
    if os.path.exists(filepath):
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    return f"Arquivo {arquivo} não encontrado no conhecimento ou Obsidian."

def buscar_obsidian(termo):
    """Busca notas no Obsidian Vault que contêm o termo no título ou conteúdo."""
    resultados = []
    if not os.path.exists(OBSIDIAN_VAULT):
        return ["Obsidian Vault não encontrado."]
    for root, dirs, files in os.walk(OBSIDIAN_VAULT):
        dirs[:] = [d for d in dirs if not d.startswith('.')]
        for file in files:
            if file.endswith(".md"):
                filepath = os.path.join(root, file)
                rel_path = os.path.relpath(filepath, OBSIDIAN_VAULT)
                if termo.lower() in file.lower():
                    resultados.append(f"Título: {rel_path}")
                    continue
                try:
                    with open(filepath, "r", encoding="utf-8") as f:
                        content = f.read()
                        if termo.lower() in content.lower():
                            idx = content.lower().find(termo.lower())
                            contexto = content[max(0, idx-30):min(len(content), idx+60)].replace('\n', ' ')
                            resultados.append(f"Conteúdo: {rel_path} ('...{contexto}...')")
                except Exception:
                    pass
    return resultados[:12]
```

## Registro no banco de dados (SQLite)
```python
import sqlite3
import datetime

# Registrar relatório de célula
def registrar_celula(nome_celula, lider, presenca, visitantes, decisoes, rede):
    conn = sqlite3.connect("database/pastoral.db")
    cursor = conn.cursor()
    cursor.execute("""
        INSERT INTO relatorios_celulas
            (data_relatorio, nome_celula, lider_nome, presenca_membros, visitantes, decisoes_fe, rede)
        VALUES (?, ?, ?, ?, ?, ?, ?)
    """, (datetime.date.today().strftime("%Y-%m-%d"), nome_celula, lider, presenca, visitantes, decisoes, rede))
    conn.commit()
    conn.close()

# Registrar visitante no funil de consolidação
def registrar_visitante(nome, whatsapp, consolidador):
    conn = sqlite3.connect("database/pastoral.db")
    cursor = conn.cursor()
    cursor.execute("""
        INSERT INTO consolidacao_visitantes
            (data_visita, visitante_nome, visitante_whatsapp, consolidador_nome, contato_24h, status)
        VALUES (?, ?, ?, ?, 0, 'Pendente')
    """, (datetime.date.today().strftime("%Y-%m-%d"), nome, whatsapp, consolidador))
    conn.commit()
    conn.close()

# Atualizar status do visitante
def atualizar_consolidacao(visitante_nome, novo_status, contato_no_prazo=True):
    conn = sqlite3.connect("database/pastoral.db")
    cursor = conn.cursor()
    cursor.execute("""
        UPDATE consolidacao_visitantes
        SET status = ?, contato_24h = ?
        WHERE visitante_nome = ? AND status = 'Pendente'
    """, (novo_status, 1 if contato_no_prazo else 0, visitante_nome))
    conn.commit()
    conn.close()
```
